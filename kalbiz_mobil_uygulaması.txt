        function updateClock() {
            const now = new Date();
            const hrs = String(now.getHours()).padStart(2, '0');
            const mins = String(now.getMinutes()).padStart(2, '0');
            document.getElementById('os-time').innerText = `${hrs}:${mins}`;

        setInterval(updateClock, 1000);

        function playSynthesizedHeartbeat() {
            try {
                if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                if (audioCtx.state === 'suspended') audioCtx.resume();
                
                const now = audioCtx.currentTime;
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                
                osc.frequency.setValueAtTime(80, now);
                osc.frequency.exponentialRampToValueAtTime(30, now + 0.12);
                gain.gain.setValueAtTime(0.8, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.12);
                
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start(now);
                osc.stop(now + 0.12);
            } catch (e) {}
        }

        function triggerPartnerPulse() {
            playSynthesizedHeartbeat();
            if (navigator.vibrate) navigator.vibrate([70, 50, 40]);
            
            const heart = document.getElementById('heart-icon');
            const ripple = document.getElementById('pendant-ripple');
            
            heart.classList.add('scale-125');
            ripple.classList.remove('opacity-0');
            
            setTimeout(() => {
                heart.classList.remove('scale-125');
                ripple.classList.add('opacity-0');
            }, 300);

            showAppToast(`Canan'ın kalp atışı (${activeBpm} BPM) kolye üzerinde hissedildi!`);
        }

            document.querySelectorAll('.tab-content').forEach(tab => tab.classList.add('hidden'));
            document.getElementById(tabId).classList.remove('hidden');

            document.querySelectorAll('.nav-btn').forEach(btn => {
                btn.classList.remove('text-rose-500');
                btn.classList.add('text-gray-400');

            btnEl.classList.remove('text-gray-400');
            btnEl.classList.add('text-rose-500');

        function selectPackageType(type) {
            selectedPackage = type;
            const btnSingle = document.getElementById('pkg-btn-single');
            const btnDouble = document.getElementById('pkg-btn-double');

            if (type === 'single') {
                btnSingle.className = 'flex-1 py-2 rounded-xl text-xs font-bold transition-all bg-rose-600 text-white shadow-md';
                btnDouble.className = 'flex-1 py-2 rounded-xl text-xs font-bold transition-all bg-white/5 text-gray-400 hover:text-white';
            } else {
                btnDouble.className = 'flex-1 py-2 rounded-xl text-xs font-bold transition-all bg-rose-600 text-white shadow-md';
                btnSingle.className = 'flex-1 py-2 rounded-xl text-xs font-bold transition-all bg-white/5 text-gray-400 hover:text-white';
            }
            updateStoreDisplay();
        }

        function selectComponent(comp) {
            selectedComponent = comp;
            const btnFull = document.getElementById('comp-btn-full');
            const btnBand = document.getElementById('comp-btn-band');
            const btnNecklace = document.getElementById('comp-btn-necklace');

            [btnFull, btnBand, btnNecklace].forEach(b => {
                b.className = 'p-2 rounded-xl border border-white/10 bg-black/40 text-gray-400 hover:text-white text-center transition-all';
            });

            if (comp === 'full') {
                btnFull.className = 'p-2 rounded-xl border border-rose-500 bg-rose-500/20 text-white font-bold text-center transition-all';
            } else if (comp === 'band') {
                btnBand.className = 'p-2 rounded-xl border border-rose-500 bg-rose-500/20 text-white font-bold text-center transition-all';
            } else {
                btnNecklace.className = 'p-2 rounded-xl border border-rose-500 bg-rose-500/20 text-white font-bold text-center transition-all';
            }
            updateStoreDisplay();
        }

        function setStoreRegion(region) {
            currentRegion = region;
            const btnTr = document.getElementById('shop-reg-tr');
            const btnInt = document.getElementById('shop-reg-int');

            if (region === 'TR') {
                btnTr.className = 'px-2.5 py-1 rounded-md font-bold bg-rose-600 text-white';
                btnInt.className = 'px-2.5 py-1 rounded-md font-bold text-gray-400';
            } else {
                btnInt.className = 'px-2.5 py-1 rounded-md font-bold bg-rose-600 text-white';
                btnTr.className = 'px-2.5 py-1 rounded-md font-bold text-gray-400';
            }
            updateStoreDisplay();
        }

        function updateStoreDisplay() {
            const price = basePrices[currentRegion][selectedPackage][selectedComponent];
            const priceStr = currentRegion === 'TR' ? `${price.toLocaleString('tr-TR')} TL` : `$${price}`;
            
            document.getElementById('shop-dynamic-price').innerText = priceStr;

            const badge = document.getElementById('package-badge');
            const title = document.getElementById('shop-product-title');
            const desc = document.getElementById('shop-product-desc');

            const isDouble = selectedPackage === 'double';
            badge.innerText = isDouble ? 'Çift Özel Seti' : 'Bireysel Kullanım';

            let compTitle = '';
            let compDesc = '';

            if (selectedComponent === 'full') {
                compTitle = isDouble ? 'Kalbiz İkili Çift Ekosistemi' : 'Kalbiz Tek Kişilik Ekosistem';
                compDesc = isDouble ? '2 Adet Akıllı Kolye + 2 Adet Akıllı Bileklik' : '1 Adet Akıllı Kolye + 1 Adet Akıllı Bileklik';
            } else if (selectedComponent === 'band') {
                compTitle = isDouble ? 'Kalbiz Çift Akıllı Bileklik Seti' : 'Kalbiz Akıllı Bileklik';
                compDesc = isDouble ? '2 Adet Titreşim Senkronlu Akıllı Bileklik' : '1 Adet Titreşim Senkronlu Akıllı Bileklik';
            } else {
                compTitle = isDouble ? 'Kalbiz Çift Akıllı Kolye Seti' : 'Kalbiz Akıllı Kolye';
                compDesc = isDouble ? '2 Adet Hisse Duyarlı Akıllı Kolye' : '1 Adet Hisse Duyarlı Akıllı Kolye';
            }

            title.innerText = compTitle;
            desc.innerText = compDesc;
        }
